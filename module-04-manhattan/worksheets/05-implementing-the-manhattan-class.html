def compute_manhattan_path(position, destination): """Compute a Manhattan path from position to destination.""" path = [] current_row, current_col = position dest_row, dest_col = destination # Move south (rows increase) while current_row < dest_row: current_row = current_row + 1 path.append((current_row, current_col)) # Move north (rows decrease) while current_row > dest_row: current_row = current_row - 1 path.append((current_row, current_col)) # Move east (columns increase) while current_col < dest_col: current_col = current_col + 1 path.append((current_row, current_col)) # Move west (columns decrease) while current_col > dest_col: current_col = current_col - 1 path.append((current_row, current_col)) return path
